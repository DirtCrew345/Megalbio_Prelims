DROP TABLE IF EXISTS fact_sales;
DROP TABLE IF EXISTS dim_customer;
DROP TABLE IF EXISTS dim_branch;
DROP TABLE IF EXISTS dim_product;
DROP TABLE IF EXISTS dim_payment;

-- CREATE DIM_PRODUCT
CREATE TABLE dim_product (
    product_id INT PRIMARY KEY,
    product_name VARCHAR(100),
    product_line VARCHAR(100),
    price NUMERIC(10,2)
);

-- CREATE DIM_CUSTOMER
CREATE TABLE dim_customer (
    customer_id INT PRIMARY KEY,
    customer_name VARCHAR(100),
    customer_type VARCHAR(50),
    gender VARCHAR(20)
);

-- CREATE DIM_BRANCH
CREATE TABLE dim_branch (
    branch_id INT PRIMARY KEY,
    branch_name VARCHAR(100),
    branch_city VARCHAR(100)
);

-- CREATE DIM_PAYMENT
CREATE TABLE dim_payment (
    payment_id INT PRIMARY KEY,
    payment_type VARCHAR(50)
);
-- CREATE FACT_SALES
CREATE TABLE fact_sales (
    invoice_id VARCHAR(20) PRIMARY KEY,
    branch_id INT NOT NULL,
    customer_id INT,
    product_id INT,
    quantity INT,
    date DATE,
    time TIME,
    payment INT,
    rating NUMERIC(3,1)
);

-- BASIC JOIN

SELECT
    fs.invoice_id,
    c.customer_name,
    p.product_name,
    p.product_line,
    b.branch_city,
    fs.quantity,
    p.price,
    fs.payment,
    fs.rating
FROM fact_sales fs

JOIN dim_customer c
    ON fs.customer_id = c.customer_id

JOIN dim_product p
    ON fs.product_id = p.product_id

JOIN dim_branch b
    ON fs.branch_id = b.branch_id;

-- AGGREGATION: COUNT

SELECT
    payment,
    COUNT(*) AS total_transactions
FROM fact_sales
GROUP BY payment
ORDER BY total_transactions DESC;

-- AGGREGATION: MIN

SELECT
    p.product_line,
    MIN(p.price) AS minimum_price
FROM dim_product p
GROUP BY p.product_line
ORDER BY minimum_price;

-- AGGREGATION: MAX

SELECT
    p.product_line,
    MAX(p.price) AS maximum_price
FROM dim_product p
GROUP BY p.product_line
ORDER BY maximum_price DESC;

-- AGGREGATION: SUM

SELECT
    p.product_line,
    SUM(fs.quantity) AS total_quantity
FROM fact_sales fs

JOIN dim_product p
    ON fs.product_id = p.product_id

GROUP BY p.product_line
ORDER BY total_quantity DESC;

-- AGGREGATION: AVG
SELECT
    p.product_line,
    ROUND(AVG(fs.rating), 2) AS average_rating
FROM fact_sales fs

JOIN dim_product p
    ON fs.product_id = p.product_id

GROUP BY p.product_line
ORDER BY average_rating DESC;



-- QUESTION 1
-- TOTAL REVENUE BY BRANCH CITY


SELECT
    b.branch_city,
    SUM(fs.quantity * p.price) AS total_revenue
FROM fact_sales fs

JOIN dim_branch b
    ON fs.branch_id = b.branch_id

JOIN dim_product p
    ON fs.product_id = p.product_id

GROUP BY b.branch_city
ORDER BY total_revenue DESC;


-- QUESTION 2
-- TOP 5 CUSTOMERS WHO SPEND THE MOST


SELECT
    c.customer_id,
    c.customer_name,
    SUM(fs.quantity * p.price) AS total_spent
FROM fact_sales fs

JOIN dim_customer c
    ON fs.customer_id = c.customer_id

JOIN dim_product p
    ON fs.product_id = p.product_id

GROUP BY
    c.customer_id,
    c.customer_name

ORDER BY total_spent DESC

LIMIT 5;


-- QUESTION 3
-- TOTAL QUANTITY SOLD BY PRODUCT LINE


SELECT
    p.product_line,
    SUM(fs.quantity) AS total_quantity_sold
FROM fact_sales fs

JOIN dim_product p
    ON fs.product_id = p.product_id

GROUP BY p.product_line

ORDER BY total_quantity_sold DESC;


-- QUESTION 4
-- DISTRIBUTION OF PAYMENT TYPE


SELECT
    dp.payment_type,
    COUNT(*) AS number_of_transactions
FROM fact_sales fs

JOIN dim_payment dp
    ON fs.payment = dp.payment_id

GROUP BY dp.payment_type

ORDER BY number_of_transactions DESC;


-- QUESTION 5
-- AVERAGE RATING PER PRODUCT LINE


SELECT
    p.product_line,
    ROUND(AVG(fs.rating), 2) AS average_rating
FROM fact_sales fs

JOIN dim_product p
    ON fs.product_id = p.product_id

GROUP BY p.product_line

ORDER BY average_rating DESC;
